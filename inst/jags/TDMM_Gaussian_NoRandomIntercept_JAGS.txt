model
{
    for (j in 1:N) {
        for (k in 1:nX) {
            betaX[j, k] <- inprod(bases.obs[j, 1:n_basis],
                                  alpha[k, 1:n_basis])
        }

        mu[j] <- inprod(betaX[j, 1:nX], X[j, 1:nX])

        y[j] ~ dnorm(mu[j], tau.e)
    }

    for (k in 1:nX) {
        alpha[k, 1:n_basis] ~ dmnorm(rep(0.00000E+00, n_basis),
                                    tau.alpha[k] * Penalty.bases[, ])
        tau.alpha[k] ~ dgamma(smoothing.shape, smoothing.rate)
    }

    tau.e ~ dgamma(residual.shape, residual.rate)
    sigma2.e <- 1 / tau.e
    sigma.e <- 1 / sqrt(tau.e)
}
