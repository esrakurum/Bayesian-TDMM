
model {
 for(j in 1:N) {
  for(k in 1:nX) {
   betaX[j,k] <- inprod(bases.obs[j,1:n_basis],alpha[k,1:n_basis])
  }
  eta[j] <- inprod(betaX[j,1:nX],X[j,1:nX])+b[subject.ID[j]]
  logit(p[j]) <- eta[j]
  y[j] ~ dbern(p[j])
 }
 for(i in 1:n.subject) { b[i] ~ dnorm(0,tau.b) }
 sigma.b ~ dunif(0,5)
 tau.b <- 1/pow(sigma.b,2)
 sigma2.b <- pow(sigma.b,2)

 for(k in 1:nX) {
  for(q in 1:n.null) { gamma[k,q] ~ dnorm(0,tau.null) }
  for(r in 1:n.pen) { theta[k,r] ~ dnorm(0,tau.alpha[k]*lambda.pen[r]) }
  for(l in 1:n_basis) {
   alpha[k,l] <- inprod(U.null[l,1:n.null],gamma[k,1:n.null])+
                 inprod(U.pen[l,1:n.pen],theta[k,1:n.pen])
  }
  tau.alpha[k] ~ dgamma(1,0.001)
 }
}
